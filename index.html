// ========================================== // ESTADO Y VARIABLES GLOBALES 🌐 // ========================================== let currentUser = null; // ========================================== // INICIALIZACIÓN Y CONFIGURACIÓN 🚀 // ========================================== document.addEventListener('DOMContentLoaded', async () => { const { data: { user } } = await supabase.auth.getUser(); if (!user) return; currentUser = user; await loadStories(); await loadFeed(); subscribeRealtime(); const postForm = document.getElementById('postForm') || document.querySelector('form'); if (postForm) { postForm.addEventListener('submit', handlePostSubmit); } }); // ========================================== // SECCIÓN DE ESTADOS / HISTORIAS ⏱️ // ========================================== async function loadStories() { const container = document.getElementById('storiesContainer'); if (!container) return; const { data: stories, error } = await supabase .from('stories') .select('*, profiles(username, avatar_url)') .gte('expires_at', new Date().toISOString()) .order('created_at', { ascending: false }); if (error) return console.error('Error cargando historias:', error); let html = `
➕

Tu estado

`; html += stories.map(story => `

${story.profiles?.username || 'Usuario'}

`).join(''); container.innerHTML = html; } async function uploadStory(file) { if (!file || !currentUser) return; const fileExt = file.name.split('.').pop(); const filePath = `${currentUser.id}/${Date.now()}.${fileExt}`; const { error: uploadError } = await supabase.storage.from('stories').upload(filePath, file); if (uploadError) return alert('Error al subir historia'); const { data } = supabase.storage.from('stories').getPublicUrl(filePath); await supabase.from('stories').insert([{ user_id: currentUser.id, media_url: data.publicUrl, type: file.type.includes('video') ? 'video' : 'image' }]); loadStories(); } // ========================================== // SECCIÓN DE INICIO / FEED Y NOTICIAS 📰 // ========================================== async function loadFeed() { const container = document.getElementById('feedContainer') || document.getElementById('chatContainer'); if (!container) return; const { data: posts, error } = await supabase .from('posts') .select('*, profiles(username, avatar_url, handle), post_likes(user_id), post_comments(*, profiles(username))') .order('created_at', { ascending: false }); if (error) return console.error('Error cargando feed:', error); container.innerHTML = posts.map(post => { const isLiked = post.post_likes?.some(like => like.user_id === currentUser.id); const likesCount = post.post_likes?.length || 0; const commentsCount = post.post_comments?.length || 0; const timeAgo = new Date(post.created_at).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }); return `

${post.profiles?.username || 'Usuario'} ${post.is_news ? '📰 NOTICIA' : ''}

@${post.profiles?.handle || 'usuario'} • ${timeAgo}
${post.content ? `

${post.content}

` : ''} ${post.image_url ? `` : ''}
${(post.post_comments || []).map(c => `
${c.profiles?.username || 'Usuario'}: ${c.content}
`).join('')}
`; }).join(''); } async function handlePostSubmit(e) { e.preventDefault(); const input = document.getElementById('postInput') || document.querySelector('input[type="text"]'); const fileInput = document.getElementById('imageInput') || document.querySelector('input[type="file"]'); if (!input && !fileInput) return; const content = input ? input.value.trim() : ''; const file = fileInput && fileInput.files ? fileInput.files[0] : null; if (!content && !file) return; let imageUrl = null; if (file) { const fileExt = file.name.split('.').pop(); const filePath = `${currentUser.id}/${Date.now()}.${fileExt}`; const { error: uploadError } = await supabase.storage.from('posts').upload(filePath, file); if (uploadError) return alert('Error al subir imagen'); const { data } = supabase.storage.from('posts').getPublicUrl(filePath); imageUrl = data.publicUrl; } await supabase.from('posts').insert([{ user_id: currentUser.id, content: content || null, image_url: imageUrl }]); if (input) input.value = ''; if (fileInput) fileInput.value = ''; loadFeed(); } async function toggleLike(postId, isLiked) { if (!currentUser) return; if (isLiked) { await supabase.from('post_likes').delete().eq('post_id', postId).eq('user_id', currentUser.id); } else { await supabase.from('post_likes').insert([{ post_id: postId, user_id: currentUser.id }]); } loadFeed(); } function toggleCommentsVisibility(postId) { const el = document.getElementById(`comments-${postId}`); if (el) el.style.display = el.style.display === 'none' ? 'block' : 'none'; } async function addComment(postId) { const input = document.getElementById(`input-comment-${postId}`); if (!input || !input.value.trim()) return; await supabase.from('post_comments').insert([{ post_id: postId, user_id: currentUser.id, content: input.value.trim() }]); input.value = ''; loadFeed(); } // ========================================== // SUSCRIPCIÓN EN TIEMPO REAL ⚡ // ========================================== function subscribeRealtime() { supabase .channel('public:feed_changes') .on('postgres_changes', { event: '*', schema: 'public', table: 'posts' }, () => loadFeed()) .on('postgres_changes', { event: '*', schema: 'public', table: 'stories' }, () => loadStories()) .on('postgres_changes', { event: '*', schema: 'public', table: 'post_comments' }, () => loadFeed()) .subscribe(); }
